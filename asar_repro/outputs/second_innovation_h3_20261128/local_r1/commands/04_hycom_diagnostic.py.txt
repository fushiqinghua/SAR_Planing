from concurrent.futures import ThreadPoolExecutor
from datetime import datetime, timezone
from hashlib import sha256
import json
import os
from pathlib import Path
from urllib.request import Request, urlopen

out = Path(os.environ["ASAR_H3_HYCOM_DIAG_DIR"])
root = "https://tds.hycom.org/thredds/dodsC/ESPC-D-V02/ice/2025"
requests = {
    "dap_dds": root + ".dds",
    "dap_das": root + ".das",
    "dap_times": root + ".ascii?time",
    "provider_missing": "https://www.hycom.org/faqs/513-did-you-know-there-are-missing-days-in-the-espc-d-v02-global-1-12-analysis",
}
def fetch(item):
    key, url = item
    record = dict(url=url, started_utc=datetime.now(timezone.utc).isoformat())
    try:
        with urlopen(Request(url, headers={"Accept-Encoding": "identity"}), timeout=45) as remote:
            payload = remote.read(2 * 1024 * 1024 + 1)
            if len(payload) > 2 * 1024 * 1024:
                raise ValueError("Diagnostic response exceeded byte cap")
            (out / ("04_hycom_" + key + ".body")).write_bytes(payload)
            record.update(status=remote.status, response_url=remote.geturl(),
                          bytes=len(payload), sha256=sha256(payload).hexdigest())
    except Exception as exc:
        record.update(error=str(exc), error_type=type(exc).__name__)
    return key, record | {"finished_utc": datetime.now(timezone.utc).isoformat()}
with ThreadPoolExecutor(max_workers=2) as pool:
    evidence = dict(pool.map(fetch, requests.items()))
print(json.dumps(evidence, ensure_ascii=False, indent=2))
