from datetime import datetime, timezone
from hashlib import file_digest
import json
import os
from pathlib import Path
from urllib.request import Request, urlopen

from netCDF4 import Dataset, num2date
import numpy as np

from h3_hycom_dap import CF_EPOCH, attributes, dap_url, packed_product, time_axis, unpack

out = Path(os.environ["ASAR_H3_QA_DIR"])
target = ["2025-09-18T21:00:00+00:00"]
axis_path = out / "04_hycom_dap_times.body"
url = dap_url(target, time_axis(axis_path.read_bytes()))
with urlopen(Request(url, headers={"Accept-Encoding": "identity"}), timeout=55) as response:
    raw = response.read(256 * 1024 + 1)
    if response.status != 200 or response.geturl() != url or len(raw) > 256 * 1024:
        raise ValueError("Unverified OPeNDAP calibration response")
    source = dict(status=response.status, url=url, headers=dict(response.headers.items()))
dap = out / "07_hycom_dap_calibration.body"
with dap.open("xb") as f:
    f.write(raw)
das = out / "04_hycom_dap_das.body"
native = Path("data/raw/h3/hudson_h3_d2/local_r1/hycom/a001/hycom/ESPC-D-V02_surface_20250918.nc")
fields = packed_product(raw, 1)
attrs = attributes(das.read_bytes())
with Dataset(native) as nd:
    axes = {name: np.array_equal(fields[name], nd[name][:]) for name in ("lon", "lat")}
    want = datetime.fromisoformat(target[0])
    from_cf = num2date(nd["time"][:1], nd["time"].units, nd["time"].calendar)[0]
    axes["time"] = (np.allclose(fields["time"], (want-CF_EPOCH).total_seconds()/3600,
                                rtol=0, atol=1e-6/3600)
                    and str(from_cf) == want.strftime("%Y-%m-%d %H:%M:%S"))
    values = {}
    for name in ("ssu", "ssv", "sic"):
        var = nd[name]
        decoded = np.ma.asarray(var[:1], dtype=np.float64).filled(np.nan)
        var.set_auto_maskandscale(False)
        values[name] = dict(raw_packed_identical=np.array_equal(fields[name],np.asarray(var[:1])),
                            cf_unpacked_identical=np.array_equal(unpack(fields[name],attrs[name]),decoded,
                                                                  equal_nan=True))
def sha(path):
    with path.open("rb") as f:
        return file_digest(f, "sha256").hexdigest()
okay = all(axes.values()) and all(all(v.values()) for v in values.values())
doc = dict(status="equivalent" if okay else "blocked", utc=target[0], source=source,
           axes=axes, variables=values, compared_spatial_samples=63*45,
           files={str(path.relative_to(Path.cwd())) if path.is_absolute() else str(path): sha(path)
                  for path in (dap,das,axis_path,native)})
print(json.dumps(doc, indent=2))
if not okay:
    raise SystemExit(1)
